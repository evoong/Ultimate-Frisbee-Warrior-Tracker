# Organization-Tiered Subscriptions, Limits, and Employee Grants Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement organization-tied tiers (Free, Plus, Premium), 30-day free trials, employee grants, tier limits (members: 15/35/unlimited, stats/history: 30-day free / unlimited paid, strategies: 3 free / unlimited paid, AI chat: 5/100/unlimited per month), ad banners for free tier, and Stripe billing.

**Architecture:** Add tier schema columns to `organizations` with automated 30-day trial initialization, implement an effective tier resolution function, add backend middleware and database guards enforcing feature limits, and add frontend indicators and ad banners.

**Tech Stack:** Supabase Postgres, Node.js/Express, React, Stripe SDK.

**Spec:** `docs/superpowers/specs/2026-09-22-org-tiers-and-employee-grants-design.md`

## Global Constraints
- Organization-wide entitlement: members inherit organization tier.
- Free tier: 15 members, 30-day raw stats/history, 3 playbook strategies, 5 AI chat messages/mo, ads enabled.
- Plus tier: 35 members, unlimited stats/history, unlimited strategies, 100 AI chat messages/mo, ad-free.
- Premium tier: Unlimited members, unlimited stats/history, unlimited strategies, unlimited AI chat, ad-free.
- Employee Grant: Elevates organization to Premium indefinitely.
- Free Trial: 30-day Premium trial on organization creation.

## Review Focus
- Member limit enforcement on invite/join when moving from Free to Plus or exceeding caps.
- AI usage quota sliding reset per calendar month.
- Raw stats history date filter correctly clamping to 30 days for Free orgs.
- Playbook strategy count validation on insert.
- Employee grant override working independently of subscription state.

```sql
create type public.org_tier as enum ('free', 'plus', 'premium');
create type public.plan_source_type as enum ('stripe', 'employee_grant', 'trial');

alter table public.organizations
  add column if not exists tier public.org_tier not null default 'free',
  add column if not exists trial_ends_at timestamptz not null default (now() + interval '30 days'),
  add column if not exists plan_source public.plan_source_type not null default 'trial',
  add column if not exists is_employee_granted boolean not null default false,
  add column if not exists stripe_customer_id text,
  add column if not exists stripe_subscription_id text;

create table if not exists public.organization_subscriptions (
  id bigint generated by default as identity primary key,
  organization_id bigint not null references public.organizations(id) on delete cascade,
  stripe_subscription_id text not null,
  status text not null,
  current_period_end timestamptz,
  created_at timestamptz not null default now()
);

create table if not exists public.ai_usage_logs (
  id bigint generated by default as identity primary key,
  organization_id bigint not null references public.organizations(id) on delete cascade,
  month_key text not null, -- 'YYYY-MM'
  message_count integer not null default 0,
  unique (organization_id, month_key)
);

create or replace function public.effective_tier(p_org_id bigint)
returns public.org_tier
language sql
security definer
stable
as $$
  select case
    when o.is_employee_granted then 'premium'::public.org_tier
    when o.plan_source = 'employee_grant' then 'premium'::public.org_tier
    when o.trial_ends_at > now() then 'premium'::public.org_tier
    else o.tier
  end
  from public.organizations o
  where o.id = p_org_id;
$$;
```

- [ ] **Step 1: Write limit checking library**

```typescript
import { pool } from './db.js';

export async function getOrgEffectiveTier(orgId: number): Promise<'free' | 'plus' | 'premium'> {
  const { rows } = await pool.query('select public.effective_tier($1) as tier', [orgId]);
  return rows[0]?.tier || 'free';
}

export async function checkMemberLimit(orgId: number): Promise<boolean> {
  const tier = await getOrgEffectiveTier(orgId);
  if (tier === 'premium') return true;
  
  const limit = tier === 'plus' ? 35 : 15;
  const { rows } = await pool.query(
    'select count(*) as cnt from public.team_members m join public.teams t on t.id = m.team_id where t.organization_id = $1',
    [orgId]
  );
  const count = parseInt(rows[0]?.cnt || '0', 10);
  return count < limit;
}

export async function checkAiMessageLimit(orgId: number): Promise<boolean> {
  const tier = await getOrgEffectiveTier(orgId);
  if (tier === 'premium') return true;

  const limit = tier === 'plus' ? 100 : 5;
  const monthKey = new Date().toISOString().slice(0, 7); // 'YYYY-MM'

  const { rows } = await pool.query(
    'select message_count from public.ai_usage_logs where organization_id = $1 and month_key = $2',
    [orgId, monthKey]
  );
  const count = parseInt(rows[0]?.message_count || '0', 10);
  return count < limit;
}

export async function incrementAiUsage(orgId: number): Promise<void> {
  const monthKey = new Date().toISOString().slice(0, 7);
  await pool.query(
    `insert into public.ai_usage_logs (organization_id, month_key, message_count)
     values ($1, $2, 1)
     on conflict (organization_id, month_key)
     do update set message_count = ai_usage_logs.message_count + 1`,
    [orgId, monthKey]
  );
}
```
